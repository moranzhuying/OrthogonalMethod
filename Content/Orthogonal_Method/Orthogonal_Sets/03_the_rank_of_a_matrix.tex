\section{矩阵的秩}

\begin{theorem}{基于正交化过程的矩阵秩计算方法}{}
	设$\boldsymbol{A}\in\mathbf{M}_{m,n}\left(\R\right),\boldsymbol{U}\in\GL_{n}\left(\R\right)$.以$\left(\col_{i}\left(\boldsymbol{U}\right)\right)_{i=1}^{n}$为初始基,对$\left(\col\left(\boldsymbol{A}\right)\right)_{i=1}^{n}$作算法(\ref{algo:orthogonal-decomposition-and-orthogonal-complement-algorithm-for-vector-subspaces}),则$\rank\left(\boldsymbol{A}\right)$是成功找到主元的个数.
\end{theorem}
